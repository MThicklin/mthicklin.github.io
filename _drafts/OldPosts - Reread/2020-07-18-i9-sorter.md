<p class="has-text-align-center">-OR-Because I'm a glutton for punishment, I took another request from HR...

Like so many "headache project" stories, this one starts with a request from HR.  They had a slowly creeping problem with the amount of I-9 forms they had to track.  When people are hired, HR makes fills out an I-9 form to verify your legal status and work authorization.  They have to hold these forms for 3 years after hiring or 1 year after being terminated, which ever is later.  Fast forward 20 years of not maintaining these forms and cleaning up manually is now an unwieldy task.  However, They did keep an excel document with everyone who had been hired and fired over the years.  Over 14000 entries spanning 20 years.  They needed a way to figure out which forms they could get rid of.

I figured, "Since the info is in excel, I could make a macro to knock this out."  Thankfully I have friends who could see the signs of old age and dementia setting in and immediately had me sectioned.  Even the other patients at the asylum asked why I would want to do something so crazy.  So, I figured I could expand my experience into C#.  

Let's set up the user story:We want to read an excel master list, use the dates to calculate which I-9 forms can be dropped, then print them out.  I-9 forms need to stay on hand at least 3 years after someone has been hired or 1 year after they have been fired, whichever is later.

Example:Someone gets hired on 4/5/2000, the I9 form will be held until 4/5/2003.  But let say it doesn't work out and this person is terminated on 4/6/2000, We still hold the form until 4/5/2001.  If the person stayed on until 2/5/2003, then we hold the form until 2/5/2004 because this date is later.  Wacky, right?!

using System.Runtime.InteropServices;
using System.Windows.Forms;
using Excel = Microsoft.Office.Interop.Excel;
namespace I9_sorter
{
public partial class i9_sorter : Form

public i9_sorter()

}private void Form1_Load(object sender, EventArgs e)
    {
        OpenFileDialog op1 = new OpenFileDialog();
        op1.Multiselect = false;
        op1.Filter = "Old Excel|*.xls|New Excel|*.xlsx";
        if (op1.ShowDialog() == DialogResult.OK)
        {
            Excel.Application xlApp = new Excel.Application();
            Excel.Workbook xlWorkbook = xlApp.Workbooks.Add(op1.FileName);
            Excel.Worksheet xlWorksheet = xlWorkbook.Sheets[3];
            Excel.Range xlRange = xlWorksheet.UsedRange;

            int rowCount = xlRange.Rows.Count;
            int colCount = xlRange.Columns.Count;

            for (int i = 2; i <= rowCount; i++)
            {
                string missing = " is missing a date on line: " + i;

                string hireName = xlWorksheet.Cells[i, 5].Value2;

                var hireCheck = xlWorksheet.Cells[i, 27].Value2;
                var termCheck = xlWorksheet.Cells[i, 89].Value2;

                DateTime hireDate;
                DateTime termDate;

                if (hireCheck == null || termCheck == null)
                {
                    dynamic fullRow = new ListViewItem(new[] { hireName, missing });
                    results.Items.Add(fullRow);
                    continue;
                }
                    else
                    {
                        hireDate = DateTime.FromOADate(xlWorksheet.Cells[i, 27].Value2);
                        termDate = DateTime.FromOADate(xlWorksheet.Cells[i, 89].Value2);
                    }

                var hireYear = hireDate.AddYears(3);
                var termYear = termDate.AddYears(1);

                if (hireYear > termYear)
                {
                    var fullRow = new ListViewItem (new[] { hireName, hireYear.ToShortDateString() } );
                    results.Items.Add(fullRow);
                }
                else
                {
                    var fullRow = new ListViewItem(new[] { hireName, termYear.ToShortDateString() });
                    results.Items.Add(fullRow);
                }
            }
            MessageBox.Show(op1.FileName + " Calculated");
        }
        else
        {
            MessageBox.Show("File Upload cancelled.");
        }
    }
    private void excel_Click(object sender, EventArgs e)
    {
        string path = Environment.GetFolderPath(Environment.SpecialFolder.Desktop);
        string file = "\\I9 Date List.xlsx";
        string save = path + file;
        object misValue = System.Reflection.Missing.Value;

        Excel.Application oXL;
        Excel._Workbook oWB;
        Excel._Worksheet oSheet;

        try
        {
            oXL = new Excel.Application();
            oWB = oXL.Workbooks.Add(1);
            oSheet = oWB.ActiveSheet;

            int row = 1;
            int col = 1;

        foreach (ListViewItem i in results.Items)
        {
                row++;
                foreach (ListViewItem.ListViewSubItem lvs in i.SubItems)
                {
                    oSheet.Cells[row, col] = lvs.Text;

                    col++;
                    if (col == 3)
                    {
                        col = 1;
                    }
                }
        }
            oWB.SaveAs(save);
            oWB.Close(true);
            oXL.Quit();

            MessageBox.Show("I9 list saved at "+save);

            Marshal.ReleaseComObject(oSheet);
            Marshal.ReleaseComObject(oWB);
            Marshal.ReleaseComObject(oXL);
        }
        catch (Exception theException)
        {
            String errorMessage;
            errorMessage = "Error: ";
            errorMessage = String.Concat(errorMessage, theException.Message);
            errorMessage = String.Concat(errorMessage, " Line: ");
            errorMessage = String.Concat(errorMessage, theException.Source);

            MessageBox.Show(errorMessage, "Error");
        }
    }
